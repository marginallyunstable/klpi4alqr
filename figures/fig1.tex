% #1 file stem, #2 x column, #3 colour, #4 marker, #5 path-name prefix, #6 end label, #7 label anchor
\newcommand{\bandcurve}[7]{%
  \addplot[name path=lo#5, draw=none, forget plot] table[x=#2, y=lo] {\datadir/#1.dat};
  \addplot[name path=hi#5, draw=none, forget plot] table[x=#2, y=hi] {\datadir/#1.dat};
  \addplot[band=#3, forget plot] fill between[of=lo#5 and hi#5];
  \addplot[#3, line width=1.0pt, mark=#4, mark size=1.5pt, mark options={solid, fill=white},
    forget plot] table[x=#2, y=r] {\datadir/#1.dat}
    node[pos=1, anchor=#7, font=\scriptsize, text=#3, inner sep=2pt] {#6};
}
% #1 test, #2 colour, #3 marker options
\newcommand{\rndclass}[3]{%
  \addplot[only marks, #3, error bars/.cd, y dir=both, y explicit,
    error bar style={line width=0.6pt, #2}, error mark options={rotate=90, mark size=1.2pt, line width=0.6pt, #2}]
    table[x expr={(#1) ? \thisrow{spread} : nan}, y=r,
      y error plus expr=\thisrow{hi}-\thisrow{r}, y error minus expr=\thisrow{r}-\thisrow{lo}]
    {\datadir/g1_random.dat};
}
